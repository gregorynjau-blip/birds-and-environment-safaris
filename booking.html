function sendBooking(event){ event.preventDefault(); const source = event.target; const method = event.submitter ? event.submitter.value : ''; const name = (source.querySelector('#name').value || '').trim(); const whatsapp = (source.querySelector('#whatsapp').value || '').trim(); const email = (source.querySelector('#email').value || '').trim(); if(method === 'whatsapp' && !whatsapp){ alert('Please enter your WhatsApp number.'); return; } if(method === 'email' && !email){ alert('Please enter your email address.'); return; } const msg = [ 'Safari Booking Request', 'Name: ' + name, 'WhatsApp: ' + (whatsapp || 'Not provided'), 'Email: ' + (email || 'Not provided'), 'Destination: ' + source.querySelector('#destination').value, 'Travel date: ' + source.querySelector('#date').value, 'Travellers: ' + source.querySelector('#travellers').value, 'Message: ' + (source.querySelector('#message').value || 'None') ].join('\n'); if(method === 'email'){ const form = document.createElement('form'); form.method = 'POST'; form.action = 'https://formspree.io/f/xkjwpdql'; form.style.display = 'none'; const fields = { name: name, whatsapp: whatsapp || 'Not provided', email: email, destination: source.querySelector('#destination').value, accommodation: source.querySelector('#accommodation').value, date: source.querySelector('#date').value, travellers: source.querySelector('#travellers').value, message: source.querySelector('#message').value || 'None', _subject: 'Safari Booking Request - ' + source.querySelector('#destination').value, _replyto: email }; Object.keys(fields).forEach(function(key){ const input = document.createElement('input'); input.type = 'hidden'; input.name = key; input.value = fields[key]; form.appendChild(input); }); document.body.appendChild(form); form.submit(); return; } window.open( 'https://wa.me/255716108475?text=' + encodeURIComponent(msg), '_blank' ); }
